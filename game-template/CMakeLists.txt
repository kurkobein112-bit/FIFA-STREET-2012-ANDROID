cmake_minimum_required(VERSION 3.25)
project(fifastreet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(generated/rexglue.cmake)
if(NOT ANDROID)
    message(FATAL_ERROR "This application target requires Android ARM64")
endif()
add_library(fifastreet SHARED src/main.cpp)
rexglue_setup_target(fifastreet GPU_PLUGINS xenos)
target_include_directories(fifastreet PRIVATE src)
if(TARGET imgui::imgui)
    target_include_directories(fifastreet PRIVATE $<TARGET_PROPERTY:imgui::imgui,INTERFACE_INCLUDE_DIRECTORIES>)
endif()
target_compile_options(fifastreet PRIVATE -O2)
if(TARGET fifastreet_recomp)
    target_compile_options(fifastreet_recomp PRIVATE -O2)
endif()
if(TARGET fifastreet_fifadllzf_xex)
    target_compile_options(fifastreet_fifadllzf_xex PRIVATE -O2)
    target_sources(fifastreet_fifadllzf_xex PRIVATE src/stubs.cpp src/native_libc_hooks.cpp src/resolution_hooks.cpp)
endif()
if(TARGET fifastreet_FootballCompEngzf_xex)
    target_compile_options(fifastreet_FootballCompEngzf_xex PRIVATE -O2)
endif()
# The dispatch registration table does not benefit from expensive optimization.
foreach(module default fifadllzf_xex FootballCompEngzf_xex)
    set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/generated/${module}/fifastreet_register.cpp"
        PROPERTIES COMPILE_OPTIONS "-O0" SKIP_PRECOMPILE_HEADERS ON)
endforeach()
